@name EntextraCore KeyValue Tests
@trigger none
@strict

#include "libs/jest"

#ifdef entity:setKeyValue(string, string)
#ifdef propSpawn(string, vector, angle, number)

function entity spawnBarrel() {
    return propSpawn("models/props_c17/oildrum001.mdl", entity():pos() + vec(0, 0, 48), ang(), 1)
}

function string uniqueKey(Label:string) {
    return "eeck_" + entity():id() + "_" + entity():creationID() + "_" + Label
}

describe("entity key-values", function() {
    beforeAll(function() {
        propSpawnUndo(0)
        propSpawnEffect(0)
    })

    it("sets, reads, and removes a key", function() {
        if (!propCanCreate()) {
            fail("propCanCreate is 0")
            return
        }

        local Ent = spawnBarrel()
        local Key = uniqueKey("color")

        expect(Ent:getKeyValue(Key)):toBe("")
        expect(Ent:getKeyValues()):toHaveLength(0)

        Ent:setKeyValue(Key, "red")
        expect(Ent:getKeyValue(Key)):toBe("red")
        expect(Ent:getKeyValues()):toHaveLength(1)
        expect(Ent:getKeyValues()[Key, string]):toBe("red")

        Ent:setKeyValue(Key, "blue")
        expect(Ent:getKeyValue(Key)):toBe("blue")
        expect(Ent:getKeyValues()):toHaveLength(1)

        Ent:setKeyValue(uniqueKey("team"), "alpha")
        expect(Ent:getKeyValues()):toHaveLength(2)

        Ent:removeKeyValue(Key)
        expect(Ent:getKeyValue(Key)):toBe("")
        expect(Ent:getKeyValues()):toHaveLength(1)

        Ent:removeKeyValue("missing")
        expect(Ent:getKeyValues()):toHaveLength(1)

        Ent:propDelete()
    })

    it("filters the world list with getEntitiesByKeyValue", function() {
        if (!propCanCreate()) {
            fail("propCanCreate is 0")
            return
        }

        local Ent = spawnBarrel()
        local Key = uniqueKey("world")

        Ent:setKeyValue(Key, "yes")
        expect(getEntitiesByKeyValue(Key, "yes")):toContain(Ent)
        expect(getEntitiesByKeyValue(Key, "no")):not():toContain(Ent)

        Ent:removeKeyValue(Key)
        expect(getEntitiesByKeyValue(Key, "yes")):not():toContain(Ent)

        Ent:propDelete()
    })
})

describe("array key-values", function() {
    beforeAll(function() {
        propSpawnUndo(0)
        propSpawnEffect(0)
    })

    it("sets, filters, and removes keys on an array", function() {
        if (!propCanCreate()) {
            fail("propCanCreate is 0")
            return
        }

        local First = spawnBarrel()
        local Second = spawnBarrel()
        local Key = uniqueKey("array")
        local Ents = array(First, Second)

        Ents:setKeyValue(Key, "on")
        expect(First:getKeyValue(Key)):toBe("on")
        expect(Second:getKeyValue(Key)):toBe("on")

        local Found = Ents:haveKeyValue(Key, "on")
        expect(Found):toHaveLength(2)
        expect(Found):toContain(First)
        expect(Found):toContain(Second)

        First:setKeyValue(Key, "off")
        Found = Ents:haveKeyValue(Key, "on")
        expect(Found):toHaveLength(1)
        expect(Found[1, entity]):toBe(Second)

        Ents:removeKeyValue(Key)
        expect(Second:getKeyValue(Key)):toBe("")
        expect(Ents:haveKeyValue(Key, "on")):toHaveLength(0)

        First:propDelete()
        Second:propDelete()
    })

    it("skips mismatches and invalid entities", function() {
        if (!propCanCreate()) {
            fail("propCanCreate is 0")
            return
        }

        local Match = spawnBarrel()
        local Other = spawnBarrel()
        local Dead = spawnBarrel()
        local Key = uniqueKey("mixed")

        Match:setKeyValue(Key, "yes")
        Other:setKeyValue(Key, "no")
        Dead:propDelete()

        local Found = array(Match, Other, Dead):haveKeyValue(Key, "yes")
        expect(Found):toHaveLength(1)
        expect(Found[1, entity]):toBe(Match)

        Match:propDelete()
        Other:propDelete()
    })
})

describe("key-value limits", function() {
    it("rejects keys and values that are too long", function() {
        local LongKey = ""
        for (_ = 1, 65) {
            LongKey = LongKey + "k"
        }

        local LongValue = ""
        for (_ = 1, 257) {
            LongValue = LongValue + "v"
        }

        expect(function() {
            entity():setKeyValue(LongKey, "x")
        }):toThrow("Key is too long")

        expect(function() {
            entity():setKeyValue("ok", LongValue)
        }):toThrow("Value is too long")
    })

    it("enforces the per-entity key count", function() {
        if (!propCanCreate()) {
            fail("propCanCreate is 0")
            return
        }

        local Ent = spawnBarrel()
        local Prefix = uniqueKey("limit")

        for (I = 1, 32) {
            Ent:setKeyValue(Prefix + "_" + I, "v")
        }

        expect(function() {
            Ent:setKeyValue(Prefix + "_33", "v")
        }):toThrow("Entity has too many key-values")

        Ent:setKeyValue(Prefix + "_1", "updated")
        expect(Ent:getKeyValue(Prefix + "_1")):toBe("updated")

        Ent:propDelete()
    })
})

describe("invalid key-value targets", function() {
    it("throws on an invalid entity", function(Done:function) {
        if (!propCanCreate()) {
            fail("propCanCreate is 0")
            return
        }

        local Dead = spawnBarrel()
        Dead:propDelete()

        timer(0.01, function() {
            expect(function() {
                Dead:setKeyValue("k", "v")
            }, "set key-value on invalid entity"):toThrow("Invalid entity")

            expect(function() {
                Dead:removeKeyValue("k")
            }, "remove key-value on invalid entity"):toThrow("Invalid entity")

            expect(function() {
                Dead:getKeyValue("k")
            }, "get key-value on invalid entity"):toThrow("Invalid entity")

            expect(function() {
                Dead:getKeyValues()
            }, "get key-values on invalid entity"):toThrow("Invalid entity")

            Done()
        })
    })
})
